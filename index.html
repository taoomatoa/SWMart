// ======================= // 🔧 KONSTANTA // ======================= const SPREADSHEET_ID = '1rH3KICNSaD0RKTXCFCGUm7kIPCawhN3lOUilz0e1FJ4'; // ID Spreadsheet utama const SHEET_DATA = 'Transaksi'; // Nama sheet tempat menyimpan transaksi const SHEET_REF = 'Referensi'; // Nama sheet berisi data referensi (dropdown) const FOLDER_UPLOAD = '1u-Sfv8KfTgeEtbNoBil62rlWlgRjODQS'; // ID Folder Google Drive untuk upload file // ======================= // 🧩 FUNGSI BANTUAN (UTILITY) // ======================= // Mengambil spreadsheet berdasarkan ID const getSpreadsheet_ = () => SpreadsheetApp.openById(SPREADSHEET_ID); // Mengambil sheet tertentu dari spreadsheet const getSheet_ = name => getSpreadsheet_().getSheetByName(name); // ======================= // 🌐 HALAMAN HTML (Frontend Aplikasi) // ======================= // Menampilkan halaman utama saat web app dibuka function doGet() { return HtmlService.createTemplateFromFile('Index') .evaluate() .setTitle('Uang Masuk - Keluar') // Judul halaman .addMetaTag('viewport', 'width=device-width, initial-scale=1') // Responsif di HP .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); // Izinkan ditampilkan di iframe } // Menyisipkan file HTML lain (misalnya CSS, JS, partial HTML) function include(filename) { return HtmlService.createHtmlOutputFromFile(filename).getContent(); } // ======================= // 🔢 MEMBUAT ID OTOMATIS // ======================= // Mengambil ID terakhir, lalu menambah 1 untuk transaksi baru function getNextId_() { const sheet = getSheet_(SHEET_DATA); const lastRow = sheet.getLastRow(); if (lastRow < 2) return 1; // Jika belum ada data return Number(sheet.getRange(lastRow, 1).getValue()) + 1; } // ======================= // 📋 AMBIL DATA REFERENSI (untuk dropdown) // ======================= function getReferensi() { const sheet = getSheet_(SHEET_REF); const values = sheet.getDataRange().getValues(); values.shift(); // Hapus baris header const kategoriPerJenis = {}; // Simpan kategori berdasarkan jenis const metode = [...new Set(values.map(r => r[2]).filter(Boolean))]; // Ambil semua metode unik // Kelompokkan kategori berdasarkan jenis values.forEach(([jenis, kategori]) => { if (!jenis || !kategori) return; // Lewati baris kosong if (!kategoriPerJenis[jenis]) kategoriPerJenis[jenis] = []; if (!kategoriPerJenis[jenis].includes(kategori)) { kategoriPerJenis[jenis].push(kategori); } }); return { kategoriPerJenis, metode }; } // ======================= // 📑 AMBIL SEMUA TRANSAKSI // ======================= function getTransaksi() { const sheet = getSheet_(SHEET_DATA); const values = sheet.getDataRange().getValues(); if (!values || values.length < 2) return []; // Tidak ada data values.shift(); // Hapus header const tz = Session.getScriptTimeZone(); // Zona waktu script return values.map(r => ({ id: r[0], tanggal: r[1] ? Utilities.formatDate(new Date(r[1]), tz, 'yyyy-MM-dd') : '', jenis: r[2], kategori: r[3], jumlah: r[4], metode: r[5], deskripsi: r[6], file: r[7] })); } // ======================= // ➕ TAMBAH TRANSAKSI BARU // ======================= function tambahTransaksi(data) { const lock = LockService.getScriptLock(); // Kunci supaya tidak bentrok jika banyak pengguna lock.waitLock(30000); // Tunggu maksimal 30 detik try { const sheet = getSheet_(SHEET_DATA); const id = getNextId_(); // Dapatkan ID baru // Tambahkan baris baru ke sheet sheet.appendRow([ id, data.tanggal, data.jenis, data.kategori, Number(data.jumlah), data.metode, data.deskripsi, data.fileUrl || '' ]); return { success: true, message: `Transaksi #${id} berhasil disimpan` }; } finally { lock.releaseLock(); // Lepaskan kunci } } // ======================= // ✏️ UPDATE TRANSAKSI YANG ADA // ======================= function updateTransaksi(data) { const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const sheet = getSheet_(SHEET_DATA); const values = sheet.getDataRange().getValues(); for (let i = 1; i < values.length; i++) { if (values[i][0] != data.id) continue; // Cari ID yang cocok const oldFileUrl = values[i][7] || ''; // Jika ada file baru, hapus file lama di Drive if (data.fileUrl && data.fileUrl !== oldFileUrl) { const oldFileId = oldFileUrl.match(/[-\w]{25,}/); if (oldFileId && oldFileId[0]) { try { DriveApp.getFileById(oldFileId[0]).setTrashed(true); } catch (err) { Logger.log(`Gagal hapus file lama: ${err.message}`); } } } // Tentukan file mana yang akan disimpan (baru atau lama) const fileUrlToSave = data.fileUrl || oldFileUrl || ''; // Update data di baris terkait sheet.getRange(i + 1, 2, 1, 7).setValues([[ data.tanggal, data.jenis, data.kategori, Number(data.jumlah), data.metode, data.deskripsi, fileUrlToSave ]]); return { success: true, message: `Transaksi #${data.id} berhasil diperbarui` }; } return { success: false, message: 'Data tidak ditemukan' }; } finally { lock.releaseLock(); } } // ======================= // ❌ HAPUS TRANSAKSI // ======================= function hapusTransaksi(id) { const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const sheet = getSheet_(SHEET_DATA); const values = sheet.getDataRange().getValues(); for (let i = 1; i < values.length; i++) { if (values[i][0] != id) continue; const fileUrl = values[i][7]; // Hapus file di Google Drive jika ada if (fileUrl) { try { const fileIdMatch = fileUrl.match(/[-\w]{25,}/); if (fileIdMatch && fileIdMatch[0]) { DriveApp.getFileById(fileIdMatch[0]).setTrashed(true); } } catch (e) { Logger.log(`Gagal hapus file untuk transaksi #${id}: ${e.message}`); } } // Hapus baris dari sheet sheet.deleteRow(i + 1); return { success: true, message: `Transaksi #${id} dan file (jika ada) dihapus` }; } return { success: false, message: 'Data tidak ditemukan' }; } finally { lock.releaseLock(); } } // ======================= // 📤 UPLOAD FILE KE GOOGLE DRIVE // ======================= function uploadFileToDrive(formObject) { try { const folder = DriveApp.getFolderById(FOLDER_UPLOAD); // Ubah file dari base64 menjadi blob (format file) const blob = Utilities.newBlob( Utilities.base64Decode(formObject.content), formObject.type, formObject.name ); const fileName = formObject.name.toLowerCase(); const mimeType = formObject.type; const size = blob.getBytes().length; // Daftar tipe file dan ekstensi yang diperbolehkan const allowedTypes = [ 'image/jpeg', 'image/png', 'application/pdf', 'application/msword', 'application/vnd.openxmlformats-officedocument.wordprocessingml.document', 'application/vnd.ms-excel', 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet', 'application/vnd.ms-powerpoint', 'application/vnd.openxmlformats-officedocument.presentationml.presentation', 'application/zip', 'application/x-rar-compressed' ]; const allowedExt = [ '.jpg', '.jpeg', '.png', '.pdf', '.doc', '.docx', '.xls', '.xlsx', '.ppt', '.pptx', '.zip', '.rar' ]; const maxSize = 10 * 1024 * 1024; // Batas maksimal 10 MB // Validasi ekstensi dan tipe file const extValid = allowedExt.some(ext => fileName.endsWith(ext)); const typeValid = allowedTypes.includes(mimeType); if (!extValid && !typeValid) { throw new Error('Tipe file tidak diizinkan. Gunakan JPG, PNG, PDF, Word, Excel, PowerPoint, ZIP, atau RAR.'); } if (size > maxSize) { throw new Error('Ukuran file terlalu besar (maksimum 10 MB).'); } // Simpan file ke folder Drive dan buat link publik const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return { success: true, url: file.getUrl(), name: file.getName() }; } catch (e) { return { success: false, message: e.message }; } }
